ขั้นตอนฝั่งบอร์ด: ① ดึง JSON จากลิงก์ด้านบน ② หา entry ที่ id ตรงกับ ID ของตัวเอง ③ คำนวณ SHA-256("dashtron-pkg-v2|" + ID + "|" + CODE) เป็น hex ตัวพิมพ์เล็ก (ID และ CODE ตัดอักขระที่ไม่ใช่ A-Z/0-9 ออกและแปลงเป็นพิมพ์ใหญ่) ④ ถ้าเท่ากับ h → ปลดล็อค · หลังออกรหัส/ลบรหัส อาจต้องรอสูงสุดราว 5 นาทีกว่า raw.githubusercontent.com จะอัปเดต

#include <WiFiClientSecure.h>
#include <HTTPClient.h>
#include <ArduinoJson.h>   // v7
#include <mbedtls/sha256.h>

String normStr(String s){ s.toUpperCase(); String o; for(char c: s) if(isalnum((unsigned char)c)) o+=c; return o; }
String sha256Hex(const String& s){
  uint8_t d[32]; mbedtls_sha256((const unsigned char*)s.c_str(), s.length(), d, 0);
  char h[65]; for(int i=0;i<32;i++) sprintf(h+i*2,"%02x",d[i]); return String(h);
}
// boardId เช่น eFuse MAC 12 หลัก: sprintf(b,"%012llX",ESP.getEfuseMac());
bool verifyPackage(String boardId, String code, const char* url){
  String id=normStr(boardId), want=sha256Hex("dashtron-pkg-v2|"+id+"|"+normStr(code));
  WiFiClientSecure c; c.setInsecure();            // ควรใส่ root CA ถ้าต้องการความปลอดภัยเต็ม
  HTTPClient http; http.begin(c,url);
  if(http.GET()!=200){ http.end(); return false; }
  JsonDocument f; f["codes"][0]["id"]=true; f["codes"][0]["h"]=true;
  JsonDocument doc;
  DeserializationError e=deserializeJson(doc,http.getStream(),DeserializationOption::Filter(f));
  http.end(); if(e) return false;
  for(JsonObject o: doc["codes"].as<JsonArray>())
    if(id==(const char*)o["id"] && want==(const char*)o["h"]) return true;
  return false;
}
